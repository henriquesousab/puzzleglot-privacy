// Integre este match dentro de:
// service cloud.firestore { match /databases/{database}/documents { ... }
// Mantenha também o default-deny global. O Trigger Email extension pode atualizar
// os metadados de entrega usando o servidor; clientes não podem ler/editar pedidos.

match /deletion_requests/{requestId} {
  allow create: if request.auth != null
    && request.auth.token.email_verified == true
    && request.resource.data.keys().hasAll([
      'uid',
      'email',
      'requestedAt',
      'status',
      'to',
      'message'
    ])
    && request.resource.data.keys().hasOnly([
      'uid',
      'email',
      'requestedAt',
      'status',
      'to',
      'message'
    ])
    && request.resource.data.uid is string
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.email is string
    && request.resource.data.email == request.auth.token.email
    && request.resource.data.requestedAt == request.time
    && request.resource.data.status == 'pending'
    && request.resource.data.to == 'henriquesousab@gmail.com'
    && request.resource.data.message is map
    && request.resource.data.message.keys().hasAll(['subject', 'text'])
    && request.resource.data.message.keys().hasOnly(['subject', 'text'])
    && request.resource.data.message.subject == 'Solicitação de exclusão de conta - PuzzleGlot'
    && request.resource.data.message.text ==
      'Nova solicitação de exclusão de conta PuzzleGlot.\nUID: ' +
      request.auth.uid +
      '\nE-mail: ' +
      request.auth.token.email;

  allow read, update, delete: if false;
}
